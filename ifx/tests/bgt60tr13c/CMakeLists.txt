cmake_minimum_required(VERSION 3.20)
project(bgt60tr13c_protocol_tests LANGUAGES CXX)
enable_testing()
add_executable(bgt60tr13c_protocol_tests protocol_test.cpp)
target_compile_features(bgt60tr13c_protocol_tests PRIVATE cxx_std_20)
target_include_directories(bgt60tr13c_protocol_tests PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/../../..")
if(MSVC)
  target_compile_options(bgt60tr13c_protocol_tests PRIVATE /UNDEBUG)
else()
  target_compile_options(bgt60tr13c_protocol_tests PRIVATE -UNDEBUG -Wall -Wextra -Werror)
endif()
add_test(NAME bgt60tr13c_protocol COMMAND bgt60tr13c_protocol_tests)
add_executable(bgt60tr13c_acquisition_tests acquisition_test.cpp)
target_compile_features(bgt60tr13c_acquisition_tests PRIVATE cxx_std_20)
target_include_directories(bgt60tr13c_acquisition_tests PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/../../..")
if(MSVC)
  target_compile_options(bgt60tr13c_acquisition_tests PRIVATE /UNDEBUG)
else()
  target_compile_options(bgt60tr13c_acquisition_tests PRIVATE -UNDEBUG -Wall -Wextra -Werror)
endif()
add_test(NAME bgt60tr13c_acquisition COMMAND bgt60tr13c_acquisition_tests)
add_executable(bgt60tr13c_delegate_tests
  spi_delegate_test.cpp
  ../../../esphome/components/spi/spi_zephyr.cpp)
target_compile_features(bgt60tr13c_delegate_tests PRIVATE cxx_std_20)
target_compile_definitions(bgt60tr13c_delegate_tests PRIVATE USE_ZEPHYR)
target_include_directories(bgt60tr13c_delegate_tests PRIVATE stubs ../../..)
if(MSVC)
  target_compile_options(bgt60tr13c_delegate_tests PRIVATE /UNDEBUG)
else()
  target_compile_options(bgt60tr13c_delegate_tests PRIVATE -UNDEBUG -Wall -Wextra -Werror -Wno-unused-parameter)
